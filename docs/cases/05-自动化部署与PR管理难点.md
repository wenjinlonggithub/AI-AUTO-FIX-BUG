# 自动化部署与PR管理难点

## 概述

自动化部署和Pull Request管理是AI自动化BUG修复系统的最后环节，也是最容易出现问题的环节。本文档分析自动化部署和PR管理过程中的难点挑战及解决方案。

## 1. Git工作流集成挑战

### 1.1 分支策略适配

**问题描述：**
- 不同项目采用不同的Git分支策略（Git Flow、GitHub Flow、GitLab Flow等）
- 分支命名规范不统一
- 合并策略和权限控制复杂

**技术挑战：**

1. **分支策略识别**
   - 无法自动判断项目使用的分支策略
   - 不同环境的分支对应关系
   - 发布分支和特性分支的管理

2. **权限和规则适配**
   - 分支保护规则不同
   - PR审批流程差异
   - 自动合并条件各异

**解决方案：**

```javascript
class GitWorkflowAdapter {
  constructor() {
    this.strategies = {
      'git-flow': new GitFlowStrategy(),
      'github-flow': new GitHubFlowStrategy(),
      'gitlab-flow': new GitLabFlowStrategy(),
      'custom': new CustomFlowStrategy()
    };
    this.detector = new WorkflowDetector();
  }
  
  async adaptToWorkflow(projectPath, remoteUrl) {
    // 1. 检测工作流类型
    const workflowType = await this.detector.detect(projectPath, remoteUrl);
    
    // 2. 获取对应策略
    const strategy = this.strategies[workflowType.type] || this.strategies.custom;
    
    // 3. 配置策略参数
    await strategy.configure(workflowType.config);
    
    return strategy;
  }
  
  async createFixBranch(strategy, bugId, baseBranch = null) {
    // 使用策略创建修复分支
    const branchInfo = await strategy.createFixBranch(bugId, baseBranch);
    
    // 验证分支创建是否成功
    await this.validateBranchCreation(branchInfo);
    
    return branchInfo;
  }
}

class GitFlowStrategy {
  constructor() {
    this.config = {
      developBranch: 'develop',
      masterBranch: 'master',
      featurePrefix: 'feature/',
      hotfixPrefix: 'hotfix/',
      releasePrefix: 'release/'
    };
  }
  
  async configure(projectConfig) {
    // 从项目配置或.gitflow配置文件读取
    const gitflowConfig = await this.loadGitflowConfig(projectConfig.projectPath);
    if (gitflowConfig) {
      Object.assign(this.config, gitflowConfig);
    }
  }
  
  async createFixBranch(bugId, baseBranch) {
    const branchName = this.generateBranchName(bugId);
    const targetBase = baseBranch || await this.determineBaseBranch(bugId);
    
    try {
      // 切换到基础分支
      await git.checkout(targetBase);
      
      // 拉取最新代码
      await git.pull('origin', targetBase);
      
      // 创建新分支
      await git.checkout(branchName, ['-b']);
      
      return {
        branchName,
        baseBranch: targetBase,
        strategy: 'git-flow',
        targetBranch: this.determineTargetBranch(bugId)
      };
    } catch (error) {
      throw new Error(`Failed to create Git Flow branch: ${error.message}`);
    }
  }
  
  generateBranchName(bugId) {
    const severity = this.getBugSeverity(bugId);
    const prefix = severity === 'critical' ? this.config.hotfixPrefix : this.config.featurePrefix;
    return `${prefix}bug-${bugId}`;
  }
  
  async determineBaseBranch(bugId) {
    const severity = this.getBugSeverity(bugId);
    return severity === 'critical' ? this.config.masterBranch : this.config.developBranch;
  }
  
  determineTargetBranch(bugId) {
    const severity = this.getBugSeverity(bugId);
    return severity === 'critical' ? this.config.masterBranch : this.config.developBranch;
  }
}

class WorkflowDetector {
  async detect(projectPath, remoteUrl) {
    const indicators = await this.gatherIndicators(projectPath, remoteUrl);
    
    // 基于指标判断工作流类型
    if (indicators.hasGitflowConfig || indicators.hasDevelopBranch) {
      return { type: 'git-flow', config: indicators };
    }
    
    if (indicators.platform === 'github' && indicators.hasMainBranch) {
      return { type: 'github-flow', config: indicators };
    }
    
    if (indicators.platform === 'gitlab' && indicators.hasMultipleEnvBranches) {
      return { type: 'gitlab-flow', config: indicators };
    }
    
    return { type: 'custom', config: indicators };
  }
  
  async gatherIndicators(projectPath, remoteUrl) {
    const indicators = {
      platform: this.detectPlatform(remoteUrl),
      branches: await this.getBranchList(projectPath),
      hasGitflowConfig: await this.checkGitflowConfig(projectPath),
      protectedBranches: await this.getProtectedBranches(remoteUrl),
      projectPath
    };
    
    // 派生指标
    indicators.hasMainBranch = indicators.branches.includes('main');
    indicators.hasMasterBranch = indicators.branches.includes('master');
    indicators.hasDevelopBranch = indicators.branches.includes('develop');
    indicators.hasMultipleEnvBranches = this.checkEnvironmentBranches(indicators.branches);
    
    return indicators;
  }
}
```

### 1.2 冲突解决机制

**问题描述：**
- 自动合并可能产生冲突
- 冲突解决需要人工干预
- 冲突解决策略需要智能化

**解决方案：**

```javascript
class IntelligentConflictResolver {
  constructor() {
    this.conflictAnalyzer = new ConflictAnalyzer();
    this.autoResolver = new AutoConflictResolver();
    this.fallbackHandler = new ConflictFallbackHandler();
  }
  
  async resolveConflicts(conflictInfo, context) {
    // 1. 分析冲突类型和复杂度
    const analysis = await this.conflictAnalyzer.analyze(conflictInfo);
    
    // 2. 尝试自动解决
    if (analysis.autoResolvable) {
      try {
        const resolution = await this.autoResolver.resolve(conflictInfo, analysis);
        return { 
          success: true, 
          resolution, 
          method: 'automatic' 
        };
      } catch (error) {
        console.warn('Auto-resolution failed, falling back to manual process');
      }
    }
    
    // 3. 生成冲突解决建议
    const suggestions = await this.generateResolutionSuggestions(analysis, context);
    
    // 4. 创建人工干预任务
    return await this.fallbackHandler.createManualResolutionTask(
      conflictInfo, 
      analysis, 
      suggestions
    );
  }
  
  async generateResolutionSuggestions(analysis, context) {
    const suggestions = [];
    
    for (const conflict of analysis.conflicts) {
      const suggestion = await this.analyzeConflict(conflict, context);
      suggestions.push(suggestion);
    }
    
    return suggestions;
  }
  
  async analyzeConflict(conflict, context) {
    return {
      file: conflict.file,
      lines: conflict.lines,
      conflictType: conflict.type, // 'content', 'whitespace', 'naming', etc.
      recommendation: await this.getRecommendation(conflict, context),
      confidence: this.calculateConfidence(conflict),
      alternativeOptions: await this.getAlternatives(conflict, context)
    };
  }
}

class AutoConflictResolver {
  async resolve(conflictInfo, analysis) {
    const resolutions = [];
    
    for (const conflict of analysis.conflicts) {
      const resolution = await this.resolveConflict(conflict);
      resolutions.push(resolution);
    }
    
    // 应用解决方案
    await this.applyResolutions(resolutions);
    
    return resolutions;
  }
  
  async resolveConflict(conflict) {
    switch (conflict.type) {
      case 'whitespace':
        return this.resolveWhitespaceConflict(conflict);
      case 'import_order':
        return this.resolveImportOrderConflict(conflict);
      case 'formatting':
        return this.resolveFormattingConflict(conflict);
      case 'simple_addition':
        return this.resolveSimpleAdditionConflict(conflict);
      default:
        throw new Error(`Cannot auto-resolve conflict type: ${conflict.type}`);
    }
  }
  
  resolveWhitespaceConflict(conflict) {
    // 选择目标分支的空白字符约定
    const projectStyle = this.detectProjectWhitespaceStyle(conflict.file);
    return {
      action: 'use_target_whitespace',
      content: this.normalizeWhitespace(conflict.content, projectStyle)
    };
  }
  
  async resolveImportOrderConflict(conflict) {
    const imports = this.parseImports(conflict.ourContent, conflict.theirContent);
    const sortedImports = this.sortImports(imports, conflict.file);
    
    return {
      action: 'merge_and_sort_imports',
      content: this.generateImportSection(sortedImports)
    };
  }
}
```

## 2. CI/CD流水线集成

### 2.1 流水线触发和状态管理

**问题描述：**
- 不同CI/CD平台的API差异
- 流水线状态监控复杂
- 构建失败的处理策略

**解决方案：**

```javascript
class CIPipelineManager {
  constructor() {
    this.providers = {
      'github-actions': new GitHubActionsProvider(),
      'gitlab-ci': new GitLabCIProvider(),
      'jenkins': new JenkinsProvider(),
      'azure-pipelines': new AzurePipelinesProvider()
    };
  }
  
  async detectCIProvider(projectPath, remoteUrl) {
    const detectors = [
      () => this.checkGitHubActions(projectPath),
      () => this.checkGitLabCI(projectPath),
      () => this.checkJenkins(projectPath),
      () => this.checkAzurePipelines(projectPath)
    ];
    
    for (const detector of detectors) {
      try {
        const result = await detector();
        if (result.detected) {
          return result;
        }
      } catch (error) {
        console.warn('CI detection failed:', error);
      }
    }
    
    return { detected: false, provider: 'none' };
  }
  
  async monitorPipelineStatus(provider, pipelineId, options = {}) {
    const maxWaitTime = options.maxWaitTime || 30 * 60 * 1000; // 30分钟
    const checkInterval = options.checkInterval || 30 * 1000; // 30秒
    const startTime = Date.now();
    
    return new Promise((resolve, reject) => {
      const checkStatus = async () => {
        try {
          const status = await provider.getPipelineStatus(pipelineId);
          
          if (status.isComplete) {
            resolve(status);
            return;
          }
          
          if (Date.now() - startTime > maxWaitTime) {
            reject(new Error('Pipeline monitoring timeout'));
            return;
          }
          
          // 记录进度
          console.log(`Pipeline ${pipelineId} status: ${status.state}`);
          
          setTimeout(checkStatus, checkInterval);
        } catch (error) {
          reject(error);
        }
      };
      
      checkStatus();
    });
  }
}

class GitHubActionsProvider {
  constructor() {
    this.apiClient = new GitHubApiClient();
  }
  
  async triggerWorkflow(owner, repo, branchName, workflowId) {
    try {
      const response = await this.apiClient.post(
        `/repos/${owner}/${repo}/actions/workflows/${workflowId}/dispatches`,
        { ref: branchName }
      );
      
      return {
        success: true,
        workflowRunId: response.data.id,
        url: response.data.html_url
      };
    } catch (error) {
      throw new Error(`Failed to trigger GitHub Actions workflow: ${error.message}`);
    }
  }
  
  async getPipelineStatus(runId) {
    const response = await this.apiClient.get(`/actions/runs/${runId}`);
    const run = response.data;
    
    return {
      id: run.id,
      state: run.status,
      conclusion: run.conclusion,
      isComplete: ['completed'].includes(run.status),
      isSuccess: run.conclusion === 'success',
      startTime: run.created_at,
      endTime: run.updated_at,
      url: run.html_url,
      jobs: await this.getJobDetails(runId)
    };
  }
  
  async getJobDetails(runId) {
    const response = await this.apiClient.get(`/actions/runs/${runId}/jobs`);
    
    return response.data.jobs.map(job => ({
      id: job.id,
      name: job.name,
      status: job.status,
      conclusion: job.conclusion,
      startTime: job.started_at,
      endTime: job.completed_at,
      steps: job.steps?.map(step => ({
        name: step.name,
        status: step.status,
        conclusion: step.conclusion,
        number: step.number
      })) || []
    }));
  }
}
```

### 2.2 构建失败处理和重试机制

**解决方案：**

```javascript
class BuildFailureHandler {
  constructor() {
    this.analyzer = new BuildFailureAnalyzer();
    this.fixer = new AutoBuildFixer();
    this.notifier = new FailureNotifier();
  }
  
  async handleBuildFailure(buildInfo, context) {
    // 1. 分析失败原因
    const analysis = await this.analyzer.analyze(buildInfo);
    
    // 2. 尝试自动修复
    const autoFixResult = await this.attemptAutoFix(analysis, context);
    
    if (autoFixResult.success) {
      return {
        handled: true,
        method: 'auto-fix',
        result: autoFixResult
      };
    }
    
    // 3. 创建问题报告
    const report = await this.generateFailureReport(analysis, context);
    
    // 4. 通知相关人员
    await this.notifier.notify(report);
    
    return {
      handled: false,
      method: 'manual-intervention-required',
      report: report
    };
  }
  
  async attemptAutoFix(analysis, context) {
    const fixers = [
      new DependencyConflictFixer(),
      new TestFailureFixer(),
      new LintErrorFixer(),
      new CompilationErrorFixer()
    ];
    
    for (const fixer of fixers) {
      if (await fixer.canHandle(analysis)) {
        try {
          const result = await fixer.fix(analysis, context);
          if (result.success) {
            return result;
          }
        } catch (error) {
          console.warn(`Auto-fixer ${fixer.name} failed:`, error);
        }
      }
    }
    
    return { success: false, reason: 'No applicable auto-fixer found' };
  }
}

class TestFailureFixer {
  async canHandle(analysis) {
    return analysis.category === 'test_failure' && 
           analysis.confidence > 0.7;
  }
  
  async fix(analysis, context) {
    const fixStrategies = [];
    
    // 分析测试失败类型
    for (const failure of analysis.failures) {
      const strategy = this.determineFixStrategy(failure);
      if (strategy) {
        fixStrategies.push(strategy);
      }
    }
    
    // 应用修复策略
    const results = [];
    for (const strategy of fixStrategies) {
      try {
        const result = await this.applyStrategy(strategy, context);
        results.push(result);
      } catch (error) {
        results.push({ success: false, error: error.message });
      }
    }
    
    const successCount = results.filter(r => r.success).length;
    
    return {
      success: successCount > 0,
      appliedFixes: results,
      needsRebuild: successCount > 0
    };
  }
  
  determineFixStrategy(failure) {
    if (failure.type === 'assertion_failure') {
      return new AssertionFixStrategy(failure);
    }
    
    if (failure.type === 'timeout') {
      return new TimeoutFixStrategy(failure);
    }
    
    if (failure.type === 'mock_setup_error') {
      return new MockFixStrategy(failure);
    }
    
    return null;
  }
}
```

## 3. Pull Request质量控制

### 3.1 自动化代码审查

**问题描述：**
- PR描述生成质量参差不齐
- 代码变更影响评估困难
- 审查清单自动化程度低

**解决方案：**

```javascript
class AutomatedPRReviewer {
  constructor() {
    this.codeAnalyzer = new PRCodeAnalyzer();
    this.impactAnalyzer = new ChangeImpactAnalyzer();
    this.templateGenerator = new PRTemplateGenerator();
  }
  
  async createQualityPR(changes, bugInfo, context) {
    // 1. 分析代码变更
    const changeAnalysis = await this.codeAnalyzer.analyze(changes);
    
    // 2. 评估变更影响
    const impactAnalysis = await this.impactAnalyzer.analyze(changes, context);
    
    // 3. 生成PR描述
    const prDescription = await this.templateGenerator.generate(
      bugInfo, 
      changeAnalysis, 
      impactAnalysis
    );
    
    // 4. 创建审查清单
    const reviewChecklist = await this.generateReviewChecklist(
      changeAnalysis, 
      impactAnalysis
    );
    
    // 5. 添加标签和里程碑
    const metadata = await this.generatePRMetadata(bugInfo, impactAnalysis);
    
    return {
      title: this.generatePRTitle(bugInfo),
      description: prDescription,
      checklist: reviewChecklist,
      metadata: metadata,
      reviewers: await this.suggestReviewers(impactAnalysis, context)
    };
  }
  
  async generateReviewChecklist(changeAnalysis, impactAnalysis) {
    const checklist = [];
    
    // 基础检查项
    checklist.push(
      '- [ ] 代码遵循项目编码规范',
      '- [ ] 添加了必要的测试用例',
      '- [ ] 测试覆盖率满足要求',
      '- [ ] 文档已更新'
    );
    
    // 基于变更类型的检查项
    if (changeAnalysis.hasDataBaseChanges) {
      checklist.push(
        '- [ ] 数据库变更脚本已审查',
        '- [ ] 考虑了数据迁移和回滚',
        '- [ ] 数据库性能影响已评估'
      );
    }
    
    if (changeAnalysis.hasAPIChanges) {
      checklist.push(
        '- [ ] API接口变更向后兼容',
        '- [ ] API文档已更新',
        '- [ ] 接口测试已添加'
      );
    }
    
    if (impactAnalysis.securityRisk > 0.5) {
      checklist.push(
        '- [ ] 安全影响已评估',
        '- [ ] 敏感信息处理符合规范',
        '- [ ] 权限控制正确实现'
      );
    }
    
    if (impactAnalysis.performanceRisk > 0.5) {
      checklist.push(
        '- [ ] 性能影响已测试',
        '- [ ] 资源使用合理',
        '- [ ] 缓存策略已考虑'
      );
    }
    
    return checklist;
  }
}

class ChangeImpactAnalyzer {
  async analyze(changes, context) {
    const analysis = {
      affectedModules: [],
      riskLevel: 'low',
      securityRisk: 0,
      performanceRisk: 0,
      compatibilityRisk: 0,
      testingRequirements: [],
      deploymentConsiderations: []
    };
    
    // 分析文件变更
    for (const change of changes.files) {
      const fileAnalysis = await this.analyzeFileChange(change, context);
      this.mergeFileAnalysis(analysis, fileAnalysis);
    }
    
    // 计算整体风险等级
    analysis.riskLevel = this.calculateRiskLevel(analysis);
    
    return analysis;
  }
  
  async analyzeFileChange(change, context) {
    const fileAnalysis = {
      file: change.path,
      changeType: change.type, // 'added', 'modified', 'deleted'
      linesChanged: change.additions + change.deletions,
      affectedFunctions: [],
      dependencies: [],
      risks: []
    };
    
    if (change.type === 'modified') {
      fileAnalysis.affectedFunctions = await this.findAffectedFunctions(change);
      fileAnalysis.dependencies = await this.analyzeDependencies(change.path, context);
    }
    
    // 风险评估
    fileAnalysis.risks = await this.assessFileRisks(change, context);
    
    return fileAnalysis;
  }
  
  async assessFileRisks(change, context) {
    const risks = [];
    
    // 核心文件风险
    if (this.isCoreFile(change.path, context)) {
      risks.push({
        type: 'core_file_change',
        severity: 'high',
        description: 'Changes to core system files'
      });
    }
    
    // 配置文件风险
    if (this.isConfigFile(change.path)) {
      risks.push({
        type: 'config_change',
        severity: 'medium',
        description: 'Configuration file changes may affect system behavior'
      });
    }
    
    // 大量变更风险
    const changeVolume = change.additions + change.deletions;
    if (changeVolume > 200) {
      risks.push({
        type: 'large_change',
        severity: 'medium',
        description: `Large number of changes (${changeVolume} lines)`
      });
    }
    
    return risks;
  }
}
```

### 3.2 智能审查员分配

**解决方案：**

```javascript
class IntelligentReviewerAssigner {
  constructor() {
    this.expertise = new DeveloperExpertiseAnalyzer();
    this.availability = new DeveloperAvailabilityTracker();
    this.workload = new WorkloadBalancer();
  }
  
  async suggestReviewers(prInfo, context) {
    // 1. 分析需要的专业知识
    const requiredExpertise = await this.analyzeRequiredExpertise(prInfo);
    
    // 2. 找到有相关专业知识的开发者
    const experts = await this.expertise.findExperts(requiredExpertise, context.team);
    
    // 3. 检查可用性
    const availableExperts = await this.availability.filterAvailable(experts);
    
    // 4. 平衡工作负载
    const balancedSuggestions = await this.workload.balance(availableExperts);
    
    // 5. 生成推荐列表
    return this.generateSuggestions(balancedSuggestions, prInfo);
  }
  
  async analyzeRequiredExpertise(prInfo) {
    const expertise = new Set();
    
    // 基于文件路径分析
    for (const file of prInfo.changes.files) {
      const fileExpertise = this.mapFileToExpertise(file.path);
      fileExpertise.forEach(e => expertise.add(e));
    }
    
    // 基于变更类型分析
    if (prInfo.analysis.hasDataBaseChanges) {
      expertise.add('database');
    }
    
    if (prInfo.analysis.hasAPIChanges) {
      expertise.add('api_design');
    }
    
    if (prInfo.analysis.hasSecurityImplications) {
      expertise.add('security');
    }
    
    if (prInfo.analysis.hasPerformanceImplications) {
      expertise.add('performance');
    }
    
    return Array.from(expertise);
  }
  
  mapFileToExpertise(filePath) {
    const expertise = [];
    
    // 前端相关
    if (/\.(vue|jsx?|tsx?)$/.test(filePath)) {
      expertise.push('frontend');
    }
    
    if (/\.css$|\.scss$|\.less$/.test(filePath)) {
      expertise.push('ui_styling');
    }
    
    // 后端相关
    if (/\.java$/.test(filePath)) {
      expertise.push('backend', 'java');
    }
    
    if (/controller|service|repository/i.test(filePath)) {
      expertise.push('backend_architecture');
    }
    
    // 数据库相关
    if (/\.sql$|migration|schema/.test(filePath)) {
      expertise.push('database');
    }
    
    // 基础设施相关
    if (/dockerfile|docker-compose|\.yml$|\.yaml$/.test(filePath)) {
      expertise.push('devops');
    }
    
    // 测试相关
    if (/test|spec/.test(filePath)) {
      expertise.push('testing');
    }
    
    return expertise;
  }
}

class DeveloperExpertiseAnalyzer {
  async findExperts(requiredExpertise, team) {
    const experts = [];
    
    for (const member of team) {
      const memberExpertise = await this.analyzeMemberExpertise(member);
      const matchScore = this.calculateExpertiseMatch(requiredExpertise, memberExpertise);
      
      if (matchScore > 0.3) { // 30%以上匹配度
        experts.push({
          member,
          expertise: memberExpertise,
          matchScore,
          relevantSkills: this.getRelevantSkills(requiredExpertise, memberExpertise)
        });
      }
    }
    
    return experts.sort((a, b) => b.matchScore - a.matchScore);
  }
  
  async analyzeMemberExpertise(member) {
    // 基于历史提交分析专业领域
    const commitHistory = await this.getRecentCommits(member, 90); // 最近90天
    const fileTypes = this.analyzeCommitFileTypes(commitHistory);
    const codeAreas = this.analyzeCodeAreas(commitHistory);
    
    return {
      languages: this.extractLanguageExpertise(fileTypes),
      domains: this.extractDomainExpertise(codeAreas),
      recentActivity: commitHistory.length,
      lastCommit: commitHistory[0]?.date
    };
  }
}
```

### 4.1 渐进式部署策略

**解决方案：**

```javascript
class GradualDeploymentManager {
  constructor() {
    this.riskAssessor = new DeploymentRiskAssessor();
    this.rollbackManager = new AutoRollbackManager();
    this.monitoring = new DeploymentMonitor();
  }
  
  async planDeployment(changes, context) {
    // 1. 评估部署风险
    const riskAssessment = await this.riskAssessor.assess(changes, context);
    
    // 2. 选择部署策略
    const strategy = this.selectDeploymentStrategy(riskAssessment);
    
    // 3. 创建部署计划
    const plan = await this.createDeploymentPlan(strategy, changes, context);
    
    return plan;
  }
  
  selectDeploymentStrategy(riskAssessment) {
    if (riskAssessment.level === 'high') {
      return new CanaryDeploymentStrategy({
        initialPercentage: 1,
        increments: [5, 10, 25, 50, 100],
        rollbackThreshold: 0.01 // 1% 错误率立即回滚
      });
    }
    
    if (riskAssessment.level === 'medium') {
      return new BlueGreenDeploymentStrategy({
        warmupTime: 300, // 5分钟预热
        switchDelay: 60   // 1分钟延迟切换
      });
    }
    
    return new DirectDeploymentStrategy({
      healthCheckTimeout: 180 // 3分钟健康检查
    });
  }
  
  async executeDeployment(plan) {
    const execution = {
      id: this.generateExecutionId(),
      plan,
      status: 'started',
      steps: [],
      startTime: new Date(),
      rollbackPlan: null
    };
    
    try {
      // 准备回滚计划
      execution.rollbackPlan = await this.prepareRollbackPlan(plan);
      
      // 执行部署步骤
      for (const step of plan.steps) {
        const stepResult = await this.executeStep(step, execution);
        execution.steps.push(stepResult);
        
        if (!stepResult.success) {
          throw new Error(`Step ${step.name} failed: ${stepResult.error}`);
        }
        
        // 监控部署健康状况
        const healthCheck = await this.monitoring.checkHealth(step, execution);
        if (!healthCheck.healthy) {
          throw new Error(`Health check failed after step ${step.name}`);
        }
      }
      
      execution.status = 'completed';
      execution.endTime = new Date();
      
    } catch (error) {
      execution.status = 'failed';
      execution.error = error.message;
      
      // 自动回滚
      await this.rollbackManager.execute(execution.rollbackPlan, execution);
    }
    
    return execution;
  }
}

class CanaryDeploymentStrategy {
  constructor(options) {
    this.options = options;
  }
  
  async createPlan(changes, context) {
    const steps = [];
    
    // 初始部署到金丝雀环境
    steps.push({
      name: 'deploy_canary_initial',
      type: 'deploy',
      target: 'canary',
      percentage: this.options.initialPercentage,
      duration: 300 // 5分钟观察期
    });
    
    // 逐步增加流量
    for (const percentage of this.options.increments) {
      steps.push({
        name: `scale_canary_${percentage}`,
        type: 'scale',
        target: 'canary',
        percentage: percentage,
        duration: 600, // 10分钟观察期
        rollbackTriggers: this.getRollbackTriggers()
      });
    }
    
    // 完全切换
    steps.push({
      name: 'complete_deployment',
      type: 'switch',
      target: 'production',
      percentage: 100
    });
    
    return {
      strategy: 'canary',
      steps,
      totalDuration: this.calculateTotalDuration(steps),
      rollbackStrategy: 'immediate'
    };
  }
  
  getRollbackTriggers() {
    return [
      {
        metric: 'error_rate',
        threshold: this.options.rollbackThreshold,
        window: '5m'
      },
      {
        metric: 'response_time_p99',
        threshold: 2000, // 2秒
        window: '5m'
      },
      {
        metric: 'cpu_usage',
        threshold: 0.9, // 90%
        window: '3m'
      }
    ];
  }
}
```

### 4.2 自动回滚机制

**解决方案：**

```javascript
class AutoRollbackManager {
  constructor() {
    this.triggers = new RollbackTriggerEngine();
    this.executor = new RollbackExecutor();
    this.notifier = new RollbackNotifier();
  }
  
  async setupMonitoring(deployment, triggers) {
    const monitoring = {
      deploymentId: deployment.id,
      triggers,
      active: true,
      startTime: new Date()
    };
    
    // 启动监控循环
    this.startMonitoringLoop(monitoring);
    
    return monitoring;
  }
  
  startMonitoringLoop(monitoring) {
    const checkInterval = 30000; // 30秒检查一次
    
    const check = async () => {
      if (!monitoring.active) return;
      
      try {
        for (const trigger of monitoring.triggers) {
          const shouldRollback = await this.triggers.evaluate(trigger, monitoring);
          
          if (shouldRollback) {
            await this.initiateRollback(monitoring, trigger);
            return;
          }
        }
        
        // 继续监控
        setTimeout(check, checkInterval);
      } catch (error) {
        console.error('Rollback monitoring failed:', error);
        setTimeout(check, checkInterval);
      }
    };
    
    setTimeout(check, checkInterval);
  }
  
  async initiateRollback(monitoring, trigger) {
    console.log(`Rollback triggered by: ${trigger.metric} exceeded threshold`);
    
    monitoring.active = false;
    monitoring.rollbackReason = trigger;
    monitoring.rollbackTime = new Date();
    
    try {
      // 执行回滚
      const rollbackResult = await this.executor.execute(monitoring.deploymentId);
      
      // 通知相关人员
      await this.notifier.notifyRollback(monitoring, rollbackResult);
      
      return rollbackResult;
    } catch (error) {
      console.error('Rollback execution failed:', error);
      
      // 紧急通知
      await this.notifier.notifyRollbackFailure(monitoring, error);
      
      throw error;
    }
  }
}

class RollbackExecutor {
  async execute(deploymentId) {
    const deployment = await this.getDeploymentInfo(deploymentId);
    const rollbackPlan = deployment.rollbackPlan;
    
    const execution = {
      deploymentId,
      startTime: new Date(),
      steps: [],
      status: 'started'
    };
    
    try {
      for (const step of rollbackPlan.steps) {
        const stepResult = await this.executeRollbackStep(step, deployment);
        execution.steps.push(stepResult);
        
        if (!stepResult.success) {
          throw new Error(`Rollback step ${step.name} failed: ${stepResult.error}`);
        }
      }
      
      execution.status = 'completed';
      execution.endTime = new Date();
      
      // 验证回滚后的健康状态
      const healthCheck = await this.verifyRollbackHealth(deployment);
      execution.healthCheck = healthCheck;
      
    } catch (error) {
      execution.status = 'failed';
      execution.error = error.message;
      execution.endTime = new Date();
    }
    
    return execution;
  }
  
  async executeRollbackStep(step, deployment) {
    const startTime = new Date();
    
    try {
      let result;
      
      switch (step.type) {
        case 'restore_previous_version':
          result = await this.restorePreviousVersion(step, deployment);
          break;
        case 'scale_down_new':
          result = await this.scaleDownNewVersion(step, deployment);
          break;
        case 'redirect_traffic':
          result = await this.redirectTraffic(step, deployment);
          break;
        case 'cleanup_resources':
          result = await this.cleanupResources(step, deployment);
          break;
        default:
          throw new Error(`Unknown rollback step type: ${step.type}`);
      }
      
      return {
        step: step.name,
        success: true,
        result,
        duration: Date.now() - startTime
      };
    } catch (error) {
      return {
        step: step.name,
        success: false,
        error: error.message,
        duration: Date.now() - startTime
      };
    }
  }
}
```
